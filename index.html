function updateDynamicLabels() { const amountLabel = document.getElementById('txAmountLabel'); const dateLabel = document.getElementById('txDateLabel'); if (paymentMode === 'avista') { if (avistaSubMethod === 'boleto') { amountLabel.textContent = 'Valor do Boleto (R$)'; dateLabel.textContent = 'Vencimento do Boleto'; } else { amountLabel.textContent = 'Valor (R$)'; dateLabel.textContent = 'Data'; } } else { if (parceladoSubType === 'boleto') { amountLabel.textContent = 'Valor Total do Carnê / Boleto (R$)'; dateLabel.textContent = 'Vencimento do 1º Boleto'; } else { amountLabel.textContent = 'Valor Total da Compra (R$)'; dateLabel.textContent = 'Data da Compra'; } } } function setPaymentMode(mode) { paymentMode = mode; const btnAvista = document.getElementById('btnModeAvista'); const btnParcelado = document.getElementById('btnModeParcelado'); const avistaSub = document.getElementById('txAvistaSubContainer'); const parceladoSub = document.getElementById('txParceladoSubContainer'); if (mode === 'avista') { btnAvista.className = 'py-2 rounded-lg text-center transition cursor-pointer bg-indigo-600 text-white shadow-sm'; btnParcelado.className = 'py-2 rounded-lg text-center transition text-slate-400 hover:bg-slate-200 dark:hover:bg-slate-700 cursor-pointer'; avistaSub.classList.remove('hidden'); parceladoSub.classList.add('hidden'); } else { btnParcelado.className = 'py-2 rounded-lg text-center transition cursor-pointer bg-purple-600 text-white shadow-sm'; btnAvista.className = 'py-2 rounded-lg text-center transition text-slate-400 hover:bg-slate-200 dark:hover:bg-slate-700 cursor-pointer'; parceladoSub.classList.remove('hidden'); avistaSub.classList.add('hidden'); setParceladoSubType(parceladoSubType); } updateDynamicLabels(); toggleInvoicePaymentFields(); } function setAvistaSubMethod(method) { avistaSubMethod = method; ['cash', 'pix', 'debit', 'credit', 'boleto'].forEach(m => { const btn = document.getElementById(`btnAvista${m.charAt(0).toUpperCase() + m.slice(1)}`); if (!btn) return; if (m === method) { btn.className = 'py-1.5 rounded-lg text-center transition cursor-pointer bg-indigo-600 text-white font-bold'; } else { btn.className = 'py-1.5 rounded-lg text-center transition cursor-pointer bg-slate-200 dark:bg-slate-700 text-slate-500 dark:text-slate-300'; } }); const cardWrapper = document.getElementById('txAvistaCardWrapper'); if (method === 'credit') { cardWrapper.classList.remove('hidden'); } else { cardWrapper.classList.add('hidden'); } updateDynamicLabels(); toggleInvoicePaymentFields(); } function setParceladoSubType(subType) { parceladoSubType = subType; const btnBoleto = document.getElementById('btnParceladoBoleto'); const btnCredit = document.getElementById('btnParceladoCredit'); const cardSelectWrapper = document.getElementById('txCardSelectWrapper'); if (subType === 'boleto') { btnBoleto.className = 'py-2 rounded-lg text-center transition cursor-pointer bg-purple-600 text-white'; btnCredit.className = 'py-2 rounded-lg text-center transition cursor-pointer bg-slate-200 dark:bg-slate-700 text-slate-400'; cardSelectWrapper.classList.add('hidden'); } else { btnCredit.className = 'py-2 rounded-lg text-center transition cursor-pointer bg-purple-600 text-white'; btnBoleto.className = 'py-2 rounded-lg text-center transition cursor-pointer bg-slate-200 dark:bg-slate-700 text-slate-400'; cardSelectWrapper.classList.remove('hidden'); } updateDynamicLabels(); updateBillingNotice(); }
